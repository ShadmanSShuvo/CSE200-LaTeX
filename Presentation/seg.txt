% ---------------- SLIDE 2: THE PROBLEM ----------------
\begin{frame}{The Problem: Ranges Under Pressure}
\vspace{-0.15cm}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\small
Given an array
\[
A = [\,2,\ 1,\ 5,\ 3,\ 7,\ 4,\ 6,\ 8\,]
\]
answer a stream of interleaved operations:
\vspace{0.15cm}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \bignav{\texttt{sum(l,r)}} --- sum of $A[l..r]$
  \item \bignav{\texttt{update(i,v)}} --- assign $A[i] \gets v$
  \item \bignav{\texttt{sum(l,r)}} --- again, on the updated array
  \item \dots\ many times \dots
\end{itemize}
\vspace{0.2cm}
\begin{block}{The tension}
Queries want the array to \textit{look pre-computed}.\\
Updates keep changing it.
\end{block}
\end{column}
\begin{column}{0.45\textwidth}
\centering
\begin{tikzpicture}[scale=0.9, every node/.style={transform shape}]
  \foreach \v [count=\i from 1] in {2,1,5,3,7,4,6,8}{
    \node[arrbox] (a\i) at (\i*0.8, 0) {\v};
    \node[arridx] at (\i*0.8, -0.55) {\i};
  }
  \node[font=\tiny, text=navy!70] at (4, -0.95) {index (1-based)};

  % annotate sum(3,6)
  \draw[decorate, decoration={brace, amplitude=5pt}, gold, ultra thick]
    (2.4-0.35,0.5) -- (4.8+0.35,0.5)
    node[midway, above=6pt, font=\scriptsize\bfseries, text=darkgold]{sum(3,6) = 19};

  % update A[4] = 10
  \node[font=\scriptsize, text=warningred] at (4.55, -1.5) {update(4, 10)};
  \draw[->, warningred, thick] (4.55, -1.35) -- (a4.south);
\end{tikzpicture}
\vspace{0.3cm}

\small
\textbf{Naive answers}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Query by scanning: $O(n)$ each.
  \item $q$ queries $\Rightarrow$ $O(nq)$ total.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% ---------------- SLIDE 3: WHY EXISTING METHODS STRUGGLE ----------------
\begin{frame}{Why Existing Methods Struggle}
\small
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}l l l l@{}}
\toprule
\textbf{Approach} & \textbf{Range query} & \textbf{Point update} & \textbf{Where it breaks} \\
\midrule
Naive array         & $O(n)$    & $O(1)$    & Queries too slow when frequent. \\
Prefix sums (static)& $O(1)$    & $O(n)$    & Every update rebuilds the prefixes. \\
Fenwick tree (BIT)  & $O(\log n)$ & $O(\log n)$ & Great for prefix sums; limited node state. \\
\rowcolor{lightblue!35}
\textbf{Segment tree} & $\boldsymbol{O(\log n)}$ & $\boldsymbol{O(\log n)}$ & Flexible node state, richer merges, lazy updates. \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.35cm}
\begin{columns}[c]
\begin{column}{0.55\textwidth}
\begin{block}{Trade-off}
Structures that make \textbf{queries} cheap tend to make \textbf{updates} expensive, and vice versa.

\medskip
Segment trees strike a balance: \textbf{both} in $O(\log n)$ for a wide class of aggregations.
\end{block}
\end{column}
\begin{column}{0.45\textwidth}
\centering
\begin{tikzpicture}[scale=0.75, every node/.style={transform shape, font=\scriptsize}]

\end{column}
\end{columns}
\end{frame}

% ---------------- SLIDE 5: CORE IDEA ----------------
\begin{frame}{Core Idea --- Divide the Array into Intervals}
\small
Store a \bignav{summary} of each interval at a tree node; queries \textbf{glue} summaries together.
\vspace{-0.1cm}
\begin{center}
\begin{tikzpicture}[scale=0.95, every node/.style={transform shape, font=\scriptsize}]

  }

  \foreach \i/\v/\idx in {0/2/1, 1.5/1/2, 3/5/3, 4.5/3/4, 6/7/5, 7.5/4/6, 9/6/7, 10.5/8/8}{

    \node at (\i+0.75, 0.75) {\textbf{\v}};
    \node[font=\tiny, text=navy!70] at (\i+0.75, 0.15) {\idx};

  \node[font=\tiny, text=navy!70, anchor=west] at (12.2, 0.75) {leaves};
\end{tikzpicture}

\begin{itemize}\setlength{\itemsep}{1pt}
  \item Each node covers a contiguous interval; children split it in half.
  \item Tree height $= \lceil \log_2 n \rceil$; total nodes $\leq 4n$ (safe allocation).
  \item Any interval $[l,r]$ decomposes into at most $O(\log n)$ tree segments.
\end{itemize}
\end{frame}

% ---------------- SLIDE 7: ITEM + MERGE ABSTRACTION ----------------
\begin{frame}[fragile]{What Does a Node Store? --- \texttt{Item} + \texttt{merge}}
\small
A segment tree is best understood as a \bignav{framework}: two ingredients specialize it to any problem.
\vspace{0.15cm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{lstlisting}
struct Item {
    long long sum = 0;   // info per interval
};

Item merge(const Item& a, const Item& b) {
    return { a.sum + b.sum };
}

// Identity for empty ranges
Item identity() { return { 0 }; }
\end{lstlisting}
\end{column}
\begin{column}{0.5\textwidth}
\begin{block}{The two ingredients}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \textbf{Item}: the summary a node stores.
  \item \textbf{merge}: how to combine two adjacent summaries.
\end{itemize}
\end{block}
\vspace{0.1cm}
\begin{block}{Why this matters}
\small Swap \texttt{Item} and \texttt{merge}, and the \emph{same} algorithms work for sums, mins, maxes, gcds, XORs, matrix products, etc. \\[2pt]
The \emph{tree} is invariant; only the algebra changes.
\end{block}
\end{column}
\end{columns}
\end{frame}

% ---------------- SLIDE 8: CONSTRUCTION ----------------
\begin{frame}[fragile]{Construction --- Linear Time}
\small
Build recursively: build left child, build right child, merge them at the root.
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{lstlisting}
// t[]: segment tree array, a[]: input
// node covers interval [l, r]  (1-based)
void build(int node, int l, int r) {
    if (l == r) {
        t[node] = Item{ a[l] };  // leaf
        return;
    }
    int mid = l + (r - l) / 2;
    build(2*node,     l,     mid);
    build(2*node + 1, mid+1, r  );
    t[node] = merge(t[2*node], t[2*node+1]);
}
\end{lstlisting}
\end{column}
\begin{column}{0.45\textwidth}
\begin{block}{Recurrence}
\[
T(n) = 2\,T(n/2) + O(1) \;\;\Longrightarrow\;\; T(n) = \bigacc{O(n)}
\]
\end{block}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Roughly $2n-1$ nodes total; each node does $O(1)$ work.
  \item Storage: allocate $4n$ to be safe for non-power-of-two $n$.
  \item Height: $\lceil \log_2 n \rceil$.
\end{itemize}
\alert{Not $O(n\log n)$} --- despite the tree having $\log n$ levels, the total node count is $\Theta(n)$.
\end{column}
\end{columns}
\end{frame}

% ---------------- SLIDE 9: THREE CASES ----------------
\begin{frame}[fragile]{Range Query --- The Three Cases}
\small
When we recurse on a node covering interval $[l,r]$ with query range $[ql,qr]$:
\vspace{0.1cm}
\begin{columns}[T]
% CASE 1
\begin{column}{0.33\textwidth}
\begin{center}
\begin{tikzpicture}[scale=0.75, every node/.style={transform shape, font=\scriptsize}]
  \draw[navy, thick, fill=lightblue!20] (0,0) rectangle (4,0.6);
  \node at (2, 0.3) {node $[l,r]$};
  \draw[warningred, ultra thick, fill=warningred!30] (4.3,0) rectangle (5.5,0.6);
  \node[text=warningred] at (4.9, 0.3) {$[ql,qr]$};
\end{tikzpicture}
\end{center}
\textbf{Case 1: No overlap}\\
Return \textbf{identity}: \\
$0$ for sum, $+\infty$ for min, etc.
\end{column}
% CASE 2
\begin{column}{0.33\textwidth}
\begin{center}

% ---------------- SLIDE 11: WHY QUERY IS O(LOG N) ----------------
\begin{frame}{Why Range Query is $O(\log n)$}
\small
It is \bigwarn{not} enough to say ``the tree has height $\log n$''. A query can visit \emph{many} nodes; we must bound how many.
\vspace{0.15cm}

\begin{block}{Canonical decomposition argument}
Any range $[ql,qr]$ decomposes into a set of maximal tree intervals fully contained in $[ql,qr]$. On each level of the tree, \textbf{at most $\mathbf{2}$} maximal intervals are chosen (one near each boundary of $[ql,qr]$). Hence:
\[
\text{nodes visited} \;\leq\; 2 \cdot (\text{height}) + O(1) \;=\; O(\log n).
\]
\end{block}

\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item ``Fully inside'' nodes contribute value and \emph{stop} recursing.
  \item Only nodes \emph{straddling} a boundary of $[ql,qr]$ recurse further.
  \item Two boundaries $\Rightarrow$ at most two recursion paths $\Rightarrow$ $O(\log n)$ nodes.
\end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
\centering
\begin{tikzpicture}[scale=0.75, every node/.style={transform shape, font=\tiny}]
  \foreach \x in {0,...,15} {
    \draw[lightblue!50, thick, fill=lightblue!30] (\x*0.5, 0) rectangle (\x*0.5+0.5, 0.4);

\end{column}

\end{frame}

% ---------------- SLIDE 13: C++ IMPLEMENTATION ----------------
\begin{frame}[fragile]{Recursive C++ Implementation}
\small \emph{Convention: 1-based array \& tree indices; each node covers $[l,r]$.}
\begin{lstlisting}
struct Item { long long sum = 0; };
Item merge(const Item& a, const Item& b) { return { a.sum + b.sum }; }
Item identity() { return { 0 }; }

const int N = 1 << 17;      // e.g. n <= 1e5
Item  t[4 * N];
long long a[N + 1];

void build(int node, int l, int r) {
    if (l == r) { t[node] = { a[l] }; return; }
    int mid = l + (r - l) / 2;
    build(2*node, l, mid);  build(2*node + 1, mid + 1, r);
    t[node] = merge(t[2*node], t[2*node + 1]);
}
void update(int node, int l, int r, int i, long long v) {
    if (l == r) { t[node] = { v }; return; }
    int mid = l + (r - l) / 2;
    if (i <= mid) update(2*node,     l,     mid, i, v);
    else          update(2*node + 1, mid+1, r,   i, v);
    t[node] = merge(t[2*node], t[2*node + 1]);
}
Item query(int node, int l, int r, int ql, int qr) {
    if (qr < l || r < ql)    return identity();
    if (ql <= l && r <= qr)  return t[node];
    int mid = l + (r - l) / 2;
    return merge(query(2*node,     l,     mid, ql, qr),
                 query(2*node + 1, mid+1, r,   ql, qr));
}
\end{lstlisting}
\end{frame}

% ---------------- SLIDE 16: ASSOCIATIVITY ----------------
\begin{frame}{Why \texttt{merge} Must Be Associative}
\small
The tree groups intervals \emph{differently} on different queries. If \texttt{merge} is associative, all groupings yield the same answer.
\vspace{0.2cm}
\[
\texttt{merge}\bigl(\texttt{merge}(a, b),\, c\bigr) \;=\; \texttt{merge}\bigl(a,\, \texttt{merge}(b, c)\bigr)
\]
\vspace{-0.1cm}
\begin{center}
\begin{tikzpicture}[scale=0.9, every node/.style={transform shape, font=\scriptsize}]
  % Grouping 1
  \node[segnode] at (0,1.5) {a};  \node[segnode] at (1,1.5) {b};  \node[segnode] at (2,1.5) {c};
  \draw[gold, ultra thick] (-0.3, 1.1) rectangle (1.3, 1.9);
  \node[text=darkgold] at (0.5, 2.2) {merge $(a,b)$};
  \draw[accentgreen, ultra thick] (-0.4, 1.0) rectangle (2.3, 2.0);
  \node[text=accentgreen] at (1.0, 2.5) {then with $c$};

  \node at (3.5, 1.5) {\large $=$};

  % Grouping 2
  \node[segnode] at (5,1.5) {a};  \node[segnode] at (6,1.5) {b};  \node[segnode] at (7,1.5) {c};
  \draw[gold, ultra thick] (5.7, 1.1) rectangle (7.3, 1.9);
  \node[text=darkgold] at (6.5, 2.2) {merge $(b,c)$};
  \draw[accentgreen, ultra thick] (4.7, 1.0) rectangle (7.4, 2.0);
  \node[text=accentgreen] at (6.0, 2.5) {then with $a$};
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item Sum, min, max, gcd, xor, matrix product $-$ all associative. \checkmark
  \item Commutativity is \emph{not} required (matrix multiplication is not commutative but still works).
  \item Some segment-tree \emph{variants} (e.g.\ persistent, dynamic, merge-sort trees) relax or refine the algebraic requirements --- but ordinary range-aggregate queries rely on associativity.
\end{itemize}
\end{frame}

$\P$ Supported for suitable operations (e.g.\ range add + range sum via BIT-of-BIT or difference tricks).\;

% ---------------- SLIDE 19: RANGE UPDATE PROBLEM ----------------
\begin{frame}{Range Updates --- Why the Naive Approach Fails}
\small
Consider \texttt{add(2,7,+5)} --- add $5$ to every element in $A[2..7]$.
\vspace{0.15cm}
\begin{center}
\begin{tikzpicture}[scale=0.9, every node/.style={transform shape}]
  \foreach \v [count=\i from 1] in {2,1,5,3,7,4,6,8}{
    \node[arrbox] (a\i) at (\i*0.9, 0) {\v};
    \node[arridx] at (\i*0.9, -0.55) {\i};
  }
  \draw[warningred, ultra thick, decorate, decoration={brace, amplitude=6pt}]
    (1.8-0.4, 0.55) -- (6.3+0.4, 0.55)
    node[midway, above=6pt, font=\scriptsize\bfseries, text=warningred]{add(2,7,+5) --- 6 leaves!};
\end{tikzpicture}
\end{center}
\vspace{0.15cm}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item Updating $r-l+1$ individual leaves is $O(n)$ per range update.
  \item After the update, every ancestor in the tree must also be recomputed.
  \item For $q$ range updates, this is $O(nq)$ --- the very cost we tried to avoid.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Key insight}
If an update fully covers a node's interval, why touch its children at all? \\
$\Longrightarrow$ \bignav{Postpone} the propagation. Do it \emph{only when needed}.
\end{block}
\end{frame}

\begin{frame}{Lazy Propagation --- Postpone, Don't Cancel}
\small
Add a companion array \texttt{lazy[node]}: a pending update that logically applies to the entire interval but hasn't yet been pushed to children.
\vspace{0.15cm}

       (node) at (0,0) {node = $[l,r]$\\ sum = updated\\ \textcolor{darkgold}{lazy = pending}};
  \node[segnode, minimum width=1.8cm, minimum height=0.9cm] (lc) at (-3, -2) {left child\\ \textbf{stale}};
  \node[segnode, minimum width=1.8cm, minimum height=0.9cm] (rc) at (3, -2)  {right child\\ \textbf{stale}};
  \draw[segedge] (node) -- (lc);
  \draw[segedge] (node) -- (rc);
  \draw[->, gold, ultra thick, dashed] (node.west) to[bend right=15] node[above, text=darkgold]{push later} (lc.north);
  \draw[->, gold, ultra thick, dashed] (node.east) to[bend left=15]  node[above, text=darkgold]{push later} (rc.north);
\end{tikzpicture}
\end{center}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item When update fully covers $[l,r]$: apply it to \texttt{t[node]} and stash in \texttt{lazy[node]}. \bigacc{Done.}
  \item Before descending into a lazy node (during a later query or update): \bignav{push} the pending value down, clear the tag, then continue.
  \item Correct because we always ``pay'' the pending update the instant anyone looks below.
\end{itemize}
\end{frame}

% ---------------- SLIDE 21: LAZY CODE ----------------
\begin{frame}[fragile]{Lazy Propagation --- Range Add \& Range Sum}
\small
\begin{lstlisting}
long long t[4*N], lazy[4*N];  // lazy: pending +x to add to every leaf under node

// Push the pending increment one level down
void push(int node, int l, int r) {
    if (lazy[node] == 0) return;
    int mid = l + (r - l) / 2;
    int lc = 2*node, rc = 2*node + 1;
    t[lc]    += (long long)(mid - l + 1) * lazy[node];
    lazy[lc] += lazy[node];
    t[rc]    += (long long)(r - mid)     * lazy[node];
    lazy[rc] += lazy[node];
    lazy[node] = 0;
}
void update(int node, int l, int r, int ql, int qr, long long x) {
    if (qr < l || r < ql) return;
    if (ql <= l && r <= qr) {
        t[node]    += (long long)(r - l + 1) * x;   // <-- multiply by length!
        lazy[node] += x;
        return;
    }
    push(node, l, r);
    int mid = l + (r - l) / 2;
    update(2*node,     l,     mid, ql, qr, x);
    update(2*node + 1, mid+1, r,   ql, qr, x);
    t[node] = t[2*node] + t[2*node + 1];
}
long long query(int node, int l, int r, int ql, int qr) {
    if (qr < l || r < ql) return 0;
    if (ql <= l && r <= qr) return t[node];
    push(node, l, r);
    int mid = l + (r - l) / 2;
    return query(2*node, l, mid, ql, qr) + query(2*node + 1, mid + 1, r, ql, qr);
}
\end{lstlisting}
\end{frame}

% ---------------- SLIDE 22: LAZY CORRECTNESS & COMPLEXITY ----------------
\begin{frame}{Lazy Propagation --- Correctness \& Complexity Caveats}
\small
\begin{block}{Why multiply by segment length?}
Adding $x$ to every leaf in an interval of length $\ell$ increases the interval's sum by $\ell \cdot x$. Sum aggregates \emph{amounts}, not \emph{presences}.
\end{block}

\begin{block}{Composing pending tags}
Two nested \texttt{add} updates accumulate:
\[
\texttt{lazy[node]} \mathrel{+}= x
\]
Because addition is associative and commutative, tags stack cleanly.
\end{block}

\vspace{0.15cm}
\begin{block}{Complexity claim (be precise!)}
For \bignav{range-add + range-sum}, both update and query are $\bigacc{O(\log n)}$.\\[2pt]
\alert{This is NOT a universal $O(\log n)$ result.} It holds because:
\begin{itemize}\setlength{\itemsep}{0pt}
  \item The lazy tag ``add $x$'' can be applied to a node's aggregate in $O(1)$.
  \item Two pending tags compose in $O(1)$ (just add them).
\end{itemize}
For operations where tags don't compose in $O(1)$ --- e.g.\ ``assign the maximum'' mixed with arbitrary updates --- extra care is needed, and complexity may differ.
\end{block}
\end{frame}

% ---------------- SLIDE 24: ADVANCED VARIANTS ----------------
\begin{frame}{A Quick Tour of Advanced Variants}
\small
\begin{columns}[T]
\begin{column}{0.34\textwidth}
\begin{block}{Iterative segment tree}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Bottom-up, no recursion.
  \item Compact and cache-friendly.
  \item Same $O(\log n)$ per op.
  \item Simplest for point-update + range-query.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.34\textwidth}
\begin{block}{Persistent segment tree}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Keeps every past version.
  \item Path copying: each update creates $O(\log n)$ new nodes.
  \item Old versions remain queryable.
  \item Powers ``$k$-th smallest in range''.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.34\textwidth}
\begin{block}{Dynamic (implicit) segment tree}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Coordinate range up to $\sim 10^{18}$.
  \item Nodes allocated only when visited.
  \item $O(\log C)$ per op ($C$ = range).
  \item Alternative: coordinate compression + static tree.
\end{itemize}
\end{block}
\end{column}
\end{columns}

% ---------------- SLIDE 25: REAL APPLICATION ----------------
\begin{frame}{Real-Life Application --- Dynamic Time-Series Aggregation}
\small
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Scenario.} A resource monitor observes CPU/memory demand across time slots. It repeatedly receives:
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \bignav{add(L,R,x)} --- a new task reserves $x$ units of capacity over $[L,R]$.
  \item \bignav{query(L,R)} --- ``what is the peak (or total) demand over this window?''
\end{itemize}
\vspace{0.15cm}
Why the straightforward options don't cut it:
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Scanning $[L,R]$ per query is $O(R-L+1)$: too slow for large windows.
  \item Static prefix sums cannot absorb frequent online range updates.
  \item Recomputing everything after each update is $O(n)$.
\end{itemize}
\vspace{0.15cm}
A \bignav{segment tree with lazy propagation} handles both operations in $O(\log n)$ per request --- online and interleaved.
\end{column}
\begin{column}{0.45\textwidth}
\centering
\begin{tikzpicture}[scale=0.85, every node/.style={transform shape, font=\tiny}]
  \foreach \x/\h in {0/1.2, 0.5/1.6, 1/2.1, 1.5/2.4, 2/2.8, 2.5/2.4, 3/2.7, 3.5/2.9, 4/2.5, 4.5/2.2, 5/1.8, 5.5/1.4}{
    \fill[deepblue!70] (\x, 0) rectangle (\x+0.4, \h);
  }
  \draw[->, navy, thick] (-0.2, 0) -- (6.1, 0) node[right, font=\scriptsize]{time};
  \draw[->, navy, thick] (-0.2, 0) -- (-0.2, 3.3) node[above, font=\scriptsize]{load};
  \draw[gold, ultra thick, decorate, decoration={brace, amplitude=5pt}]
    (1, -0.15) -- (3.5, -0.15) node[midway, below=6pt, font=\scriptsize\bfseries, text=darkgold]{query [L,R]};
  \draw[warningred, ultra thick, ->] (2, 3.4) -- (2, 2.9) node[above=6pt, font=\scriptsize, text=warningred, right=-2pt]{add $+x$};
\end{tikzpicture}
\vspace{0.3cm}

\begin{block}{Caveat}
Segment trees are a natural fit --- not the only possibility. In practice one may also consider interval trees, Fenwick variants, or specialized indexes depending on exact query mix.
\end{block}
\end{column}
\end{columns}
\end{frame}

% ---------------- SLIDE 26: SEG TREE vs FENWICK ----------------
\begin{frame}{Segment Tree vs Fenwick Tree}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Pick a Fenwick tree when\ldots}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item You need prefix/range \textbf{sums} (or invertible aggregates).
  \item \textbf{Point} updates dominate.
  \item You want a \textbf{tiny}, cache-friendly implementation.
  \item Memory budget is very tight.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Pick a segment tree when\ldots}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item You need \textbf{arbitrary} associative aggregates (min, max, gcd, \dots).
  \item Nodes must store \textbf{richer state} (multiple fields).
  \item You need \textbf{range updates} with lazy propagation.
  \item You need advanced variants: \textbf{persistent, dynamic, merge-sort tree}, \dots
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}[every node/.style={font=\scriptsize}]
  \node[draw=warningred, fill=warningred!10, rounded corners=3pt,
        text=warningred, inner sep=6pt, align=center] {%
    \textbf{Common misconception:} ``Segment trees are always faster / always better.''\\
    Reality: they trade constants and simplicity for flexibility. Both structures are excellent when used in their sweet spot.
  };
\end{tikzpicture}
\end{center}
\end{frame}

\begin{itemize}\setlength{\itemsep}{2pt}

  \item Build $O(n)$; query and point update $O(\log n)$ per operation.
